   288→function HighlightedText({
   289→  text,
   290→  findings,
   291→  actionsBy,
   292→  activeKey,
   293→  onSelect,
   294→}: {
   295→  text: string
   296→  findings: Finding[]
   297→  actionsBy: Map<string, FindingActionDTO>
   298→  activeKey: string | null
   299→  onSelect: (key: string) => void
   300→}) {
   301→  const segments = useMemo(() => {
   302→    const marks: { start: number; end: number; finding: Finding }[] = []
   303→    const lower = text.toLowerCase()
   304→    for (const f of findings) {
   305→      if (!f.quote) continue
   306→      const idx = lower.indexOf(f.quote.toLowerCase())
   307→      if (idx === -1) continue
   308→      const end = idx + f.quote.length
   309→      // skip overlaps
   310→      if (marks.some((m) => idx < m.end && end > m.start)) continue
   311→      marks.push({ start: idx, end, finding: f })
   312→    }
   313→    marks.sort((a, b) => a.start - b.start)
   314→
   315→    const out: { text: string; finding?: Finding }[] = []
   316→    let pos = 0
   317→    for (const m of marks) {
   318→      if (m.start > pos) out.push({ text: text.slice(pos, m.start) })
   319→      out.push({ text: text.slice(m.start, m.end), finding: m.finding })
   320→      pos = m.end
   321→    }
   322→    if (pos < text.length) out.push({ text: text.slice(pos) })
   323→    return out
   324→  }, [text, findings])
   325→
   326→  return (
   327→    <div className="whitespace-pre-wrap text-[13.5px] leading-[1.9] text-zinc-800">
   328→      {segments.map((seg, i) =>
   329→        seg.finding ? (
   330→          <mark
   331→            key={i}
   332→            onClick={() => onSelect(seg.finding!.key)}
   333→            className={cn(
   334→              'vl-hl',
   335→              seg.finding.severity === 'critical' && 'vl-hl-critical',
   336→              seg.finding.severity === 'warning' && 'vl-hl-warning',
   337→              seg.finding.severity === 'info' && 'vl-hl-info',
   338→              activeKey === seg.finding.key && 'outline outline-2 outline-offset-1 outline-zinc-900',
   339→              actionsBy.has(seg.finding.key) && 'vl-hl-info opacity-60'
   340→            )}
   341→          >
   342→            {seg.text}
   343→            <sup className="ml-0.5 font-mono text-[9px] font-bold text-zinc-600">
   344→              {actionsBy.has(seg.finding.key) ? '✓' : '!'}
   345→            </sup>
   346→          </mark>
   347→        ) : (
   348→          <span key={i}>{seg.text}</span>
   349→        )
   350→      )}
   351→    </div>
   352→  )
   353→}

   357→function QueuedFinding({
   358→  finding,
   359→  action,
   360→  active,
   361→  auditId,
   362→  onActivate,
   363→  onChanged,
   364→}: {
   365→  finding: Finding
   366→  action: FindingActionDTO | null
   367→  active: boolean
   368→  auditId: string
   369→  onActivate: () => void
   370→  onChanged: () => Promise<void>
   371→}) {
   372→  const { role } = useApp()
   373→  const [busy, setBusy] = useState(false)
   374→  const [rewrite, setRewrite] = useState<{ rewrite: string; intentNotes: string | null } | null>(null)
   375→  const [editing, setEditing] = useState(false)
   376→  const [editText, setEditText] = useState('')
   377→  const [overrideOpen, setOverrideOpen] = useState(false)
   378→
   379→  const post = async (payload: Record<string, unknown>) => {
   380→    setBusy(true)
   381→    try {
   382→      const res = await fetch('/api/findings/action', {
   383→        method: 'POST',
   384→        headers: { 'Content-Type': 'application/json' },
   385→        body: JSON.stringify({
   386→          auditId,
   387→          findingKey: finding.key,
   388→          actor: roleIdentity(role).name,
   389→          actorRole: role,
   390→          ...payload,
   391→        }),
   392→      })
   393→      const body = await res.json()
   394→      if (!res.ok) throw new Error(body.error)
   395→      await onChanged()
   396→      return true
   397→    } catch (err) {
   398→      toast({ title: 'Action failed', description: String(err), variant: 'destructive' })
   399→      return false
   400→    } finally {
   401→      setBusy(false)
   402→    }
   403→  }
   404→
   405→  const undo = async () => {
   406→    setBusy(true)
   407→    try {
   408→      const res = await fetch('/api/findings/action', {
   409→        method: 'DELETE',
   410→        headers: { 'Content-Type': 'application/json' },
   411→        body: JSON.stringify({ auditId, findingKey: finding.key }),
   412→      })
   413→      if (!res.ok) throw new Error('undo failed')
   414→      await onChanged()
   415→    } catch {
   416→      toast({ title: 'Undo failed', variant: 'destructive' })
   417→    } finally {
   418→      setBusy(false)
   419→    }
   420→  }
   421→
   422→  return (
   423→    <motion.div
   424→      layout
   425→      onClick={onActivate}
   426→      className={cn(
   427→        'cursor-pointer rounded-xl border bg-white p-3 shadow-sm transition-colors',
   428→        active ? 'border-zinc-400 ring-1 ring-zinc-300' : 'border-zinc-200',
   429→        action && 'opacity-90'
   430→      )}
   431→    >
   432→      <OverrideDialog
   433→        open={overrideOpen}
   434→        onOpenChange={setOverrideOpen}
   435→        onSubmit={async (note) => post({ action: 'overridden', note })}
   436→      />
   437→      <div className="flex items-start justify-between gap-2">
   438→        <div className="min-w-0">
   439→          <div className="flex items-center gap-1.5">
   440→            <SeverityChip severity={finding.severity} />
   441→            <span className="font-mono text-[10px] font-bold text-zinc-400">{finding.key}</span>
   442→          </div>
   443→          <p className="mt-1.5 text-[13px] font-semibold leading-snug text-zinc-900">
   444→            {finding.title}
   445→          </p>
   446→        </div>
   447→        {action && (
   448→          <span className="flex shrink-0 items-center gap-1 rounded-full bg-emerald-50 px-2 py-0.5 text-[10px] font-bold text-emerald-700">
   449→            <Check className="h-3 w-3" />
   450→            {action.action.toUpperCase()}
   451→          </span>
   452→        )}
   453→      </div>
   454→
   455→      {finding.quote && (
   456→        <p className="mt-1.5 rounded-md bg-zinc-100 px-2 py-1 font-mono text-[11px] leading-snug text-zinc-600">
   457→          “{finding.quote}”
   458→        </p>
   459→      )}
   460→
   461→      {!action ? (
   462→        <div className="mt-2.5 space-y-2" onClick={(e) => e.stopPropagation()}>
   463→          <div className="flex flex-wrap gap-1.5">
   464→            {finding.suggestion && (
   465→              <Button
   466→                size="sm"
   467→                disabled={busy}
   468→                onClick={() => post({ action: 'accepted', appliedText: finding.suggestion })}
   469→                className="h-7 bg-emerald-600 px-2.5 text-[11px] hover:bg-emerald-500"
   470→              >
   471→                <Check className="mr-1 h-3 w-3" />
   472→                Accept {finding.suggestionSource === 'ai' ? 'AI fix' : 'fix'}
   473→              </Button>
   474→            )}
   475→            {finding.quote && (
   476→              <Button
   477→                size="sm"
   478→                variant="outline"
   479→                disabled={busy}
   480→                onClick={async () => {
   481→                  setBusy(true)
   482→                  try {
   483→                    const res = await fetch('/api/rewrite', {
   484→                      method: 'POST',
   485→                      headers: { 'Content-Type': 'application/json' },
   486→                      body: JSON.stringify({ auditId, findingKey: finding.key }),
   487→                    })
   488→                    const body = await res.json()
   489→                    if (!res.ok) throw new Error(body.error)
   490→                    setRewrite(body)
   491→                  } catch (err) {
   492→                    toast({ title: 'Co-pilot unavailable', description: String(err), variant: 'destructive' })
   493→                  } finally {
   494→                    setBusy(false)
   495→                  }
   496→                }}
   497→                className="h-7 px-2.5 text-[11px]"
   498→              >
   499→                {busy ? <Loader2 className="mr-1 h-3 w-3 animate-spin" /> : <Sparkles className="mr-1 h-3 w-3" />}
   500→                Intent-preserving rewrite
   501→              </Button>
   502→            )}
   503→            <Button
   504→              size="sm"
   505→              variant="outline"
   506→              disabled={busy}
   507→              onClick={() => {
   508→                setEditText(finding.suggestion ?? finding.quote)
   509→                setEditing(true)
   510→              }}
   511→              className="h-7 px-2.5 text-[11px]"
   512→            >
   513→              <PencilLine className="mr-1 h-3 w-3" /> Edit
   514→            </Button>
   515→            <Button
   516→              size="sm"
   517→              variant="outline"
   518→              disabled={busy}
   519→              onClick={() => setOverrideOpen(true)}
   520→              className="h-7 border-amber-300 px-2.5 text-[11px] text-amber-700 hover:bg-amber-50"
   521→            >
   522→              <ShieldQuestion className="mr-1 h-3 w-3" /> Override
   523→            </Button>
   524→          </div>
   525→
   526→          {rewrite && (
   527→            <div className="rounded-lg border border-teal-200 bg-teal-50/60 p-2.5">
   528→              <p className="text-[10px] font-bold uppercase tracking-wider text-teal-700">
   529→                Co-pilot rewrite · minimal diff, intent preserved
   530→              </p>
   531→              <p className="mt-1 text-xs font-medium leading-relaxed text-teal-900">
   532→                “{rewrite.rewrite}”
   533→              </p>
   534→              {rewrite.intentNotes && (
   535→                <p className="mt-1 text-[10.5px] italic text-teal-700/80">
   536→                  {rewrite.intentNotes}
   537→                </p>
   538→              )}
   539→              <div className="mt-2 flex gap-1.5">
   540→                <Button
   541→                  size="sm"
   542→                  className="h-7 bg-teal-600 px-2.5 text-[11px] hover:bg-teal-500"
   543→                  disabled={busy}
   544→                  onClick={() => post({ action: 'accepted', appliedText: rewrite.rewrite })}
   545→                >
   546→                  <Check className="mr-1 h-3 w-3" /> Use rewrite
   547→                </Button>
   548→                <Button size="sm" variant="ghost" className="h-7 px-2 text-[11px]" onClick={() => setRewrite(null)}>
   549→                  Dismiss
   550→                </Button>
   551→              </div>
   552→            </div>
   553→          )}
   554→
   555→          {editing && (
   556→            <div className="rounded-lg border border-zinc-200 bg-zinc-50 p-2.5">
   557→              <Textarea
   558→                className="min-h-[70px] text-xs"
   559→                value={editText}
   560→                onChange={(e) => setEditText(e.target.value)}
   561→                placeholder="Write the replacement span…"
   562→              />
   563→              <div className="mt-1.5 flex gap-1.5">
   564→                <Button
   565→                  size="sm"
   566→                  className="h-7 bg-zinc-900 px-2.5 text-[11px] hover:bg-zinc-800"
   567→                  disabled={busy}
   568→                  onClick={async () => {
   569→                    if (await post({ action: 'edited', appliedText: editText })) setEditing(false)
   570→                  }}
   571→                >
   572→                  Save edit
   573→                </Button>
   574→                <Button size="sm" variant="ghost" className="h-7 px-2 text-[11px]" onClick={() => setEditing(false)}>
   575→                  Cancel
   576→                </Button>
   577→              </div>
   578→            </div>
   579→          )}
   580→        </div>
   581→      ) : (
   582→        <div className="mt-2 space-y-1.5" onClick={(e) => e.stopPropagation()}>
   583→          <div className="rounded-lg border border-zinc-100 bg-zinc-50 p-2">
   584→            <p className="text-[10px] font-bold uppercase tracking-wider text-zinc-400">
   585→              {action.action === 'overridden' ? 'Human override — original kept' : 'Replacement'}
   586→            </p>
   587→            <p className="mt-0.5 text-[11.5px] leading-snug text-zinc-700">
   588→              {action.action === 'overridden'
   589→                ? `“${action.note ?? ''}”`
   590→                : `“${finding.quote}” → “${action.appliedText}”`}
   591→            </p>
   592→          </div>
   593→          <button
   594→            onClick={undo}
   595→            disabled={busy}
   596→            className="flex items-center gap-1 text-[10.5px] font-semibold text-zinc-400 hover:text-zinc-700"
   597→          >
   598→            <Undo2 className="h-3 w-3" /> revise decision
   599→          </button>
   600→        </div>
   601→      )}
   602→    </motion.div>
   603→  )
   604→}

   662→function RemediateButton({
   663→  draftId,
   664→  auditId,
   665→  disabled,
   666→  onDone,
   667→  actor,
   668→}: {
   669→  draftId: string
   670→  auditId: string
   671→  disabled: boolean
   672→  onDone: () => Promise<void>
   673→  actor: string
   674→}) {
   675→  const [busy, setBusy] = useState(false)
   676→  const { setView } = useApp()
   677→  return (
   678→    <Button
   679→      disabled={disabled || busy}
   680→      className="bg-emerald-600 hover:bg-emerald-500"
   681→      onClick={async () => {
   682→        setBusy(true)
   683→        try {
   684→          const res = await fetch('/api/remediate', {
   685→            method: 'POST',
   686→            headers: { 'Content-Type': 'application/json' },
   687→            body: JSON.stringify({ draftId, auditId, actor }),
   688→          })
   689→          const body = await res.json()
   690→          if (!res.ok) throw new Error(body.error)
   691→          await onDone()
   692→          toast({
   693→            title: `Remediated version v${body.version} created`,
   694→            description: `${body.applied} fixes applied · ${body.overrides} override(s) preserved`,
   695→          })
   696→          setView('review')
   697→        } catch (err) {
   698→          toast({ title: 'Remediation failed', description: String(err), variant: 'destructive' })
   699→        } finally {
   700→          setBusy(false)
   701→        }
   702→      }}
   703→    >
   704→      {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <ArrowRight className="mr-2 h-4 w-4" />}
   705→      {disabled ? 'Resolve all findings first' : 'Create remediated version'}
   706→    </Button>
   707→  )
   708→}
