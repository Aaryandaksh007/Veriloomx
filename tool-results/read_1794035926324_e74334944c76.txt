   835→   835→function describeEvent(action: string, details: Record<string, unknown>): string {
   836→   836→  switch (action) {
   837→   837→    case 'AUDIT_RUN':
   838→   838→      return `6-pillar audit completed — score ${details.score}, ${details.findings} findings (${details.suppressed} suppressed)`
   839→   839→    case 'AI_CALL':
   840→   840→      return `AI call: ${String(details.purpose ?? 'inference').slice(0, 70)}`
   841→   841→    case 'OVERRIDE':
   842→   842→      return `Human override on ${details.findingKey} — “${String(details.justification ?? '').slice(0, 48)}…”`
   843→   843→    case 'STATUS_CHANGE':
   844→   844→      return `Status: ${details.from ?? '∅'} → ${details.to}`
   845→   845→    case 'CERT_ISSUED':
   846→   846→      return 'Compliance certificate issued'
   847→   847→    case 'VAULT_INIT':
   848→   848→      return String(details.message ?? 'Vault initialised')
   849→   849→    case 'VAULT_EDIT':
   850→   850→      return `Vault clause ${details.verb} — ${details.clauseId}`
   851→   851→    case 'ACCEPT':
   852→   852→      return `Finding ${details.findingKey} accepted`
   853→   853→    default:
   854→   854→      return action
   855→   855→  }
