   350→function PipelineCard({
   351→  draft,
   352→  audit,
   353→  index,
   354→}: {
   355→  draft: DraftDTO
   356→  audit?: AuditResult
   357→  index: number
   358→}) {
   359→  const { selectDraft } = useApp()
   360→
   361→  const stages = RAIL_STAGES.map((s) => ({
   362→    ...s,
   363→    done:
   364→      s.id === 'draft'
   365→        ? true
   366→        : s.id === 'audit'
   367→          ? !!audit
   368→          : s.id === 'fix'
   369→            ? fixDone(draft)
   370→            : s.id === 'review'
   371→              ? reviewDone(draft)
   372→              : draft.status === 'APPROVED',
   373→  }))
   374→  const doneCount = stages.filter((s) => s.done).length
   375→  const allDone = doneCount === stages.length
   376→  const activeIdx = allDone ? -1 : doneCount
   377→  const fillPct = allDone ? 100 : ((doneCount - 1) / (stages.length - 1)) * 100
   378→  const nextStage = allDone ? null : stages[activeIdx]
   379→  const route = routeFor(draft)
   380→  const open = () => selectDraft(draft.id, route)
   381→
   382→  const approved = draft.status === 'APPROVED'
   383→  const scoreTone =
   384→    !audit || audit.score >= 85
   385→      ? 'text-emerald-600 bg-emerald-50 border-emerald-200'
   386→      : audit.score >= 60
   387→        ? 'text-amber-600 bg-amber-50 border-amber-200'
   388→        : 'text-rose-600 bg-rose-50 border-rose-200'
   389→
   390→  return (
   391→    <motion.article
   392→      initial={{ opacity: 0, y: 16 }}
   393→      animate={{ opacity: 1, y: 0 }}
   394→      transition={{ delay: 0.06 * index, duration: 0.45, ease: [0.22, 1, 0.36, 1] }}
   395→      className="group cursor-pointer rounded-xl border border-zinc-200/80 bg-white p-4 shadow-sm transition-all duration-200 hover:-translate-y-0.5 hover:border-zinc-300 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-zinc-300"
   396→      onClick={open}
   397→      onKeyDown={(e) => {
   398→        if (e.key === 'Enter' || e.key === ' ') {
   399→          e.preventDefault()
   400→          open()
   401→        }
   402→      }}
   403→      tabIndex={0}
   404→      role="button"
   405→      aria-label={`Open ${draft.title} in ${VIEW_LABEL[route]}`}
   406→    >
   407→      {/* header row */}
   408→      <div className="flex items-start justify-between gap-3">
   409→        <div className="flex min-w-0 items-start gap-3">
   410→          <span className="mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-zinc-100 text-zinc-500 transition-colors duration-200 group-hover:bg-zinc-900 group-hover:text-white">
   411→            {draft.kind === 'image' ? (
   412→              <ImageIcon className="h-4 w-4" />
   413→            ) : (
   414→              <ScanSearch className="h-4 w-4" />
   415→            )}
   416→          </span>
   417→          <div className="min-w-0">
   418→            <h3 className="truncate text-sm font-bold leading-snug text-zinc-900">
   419→              {draft.title}
   420→            </h3>
   421→            <p className="mt-0.5 truncate text-[11px] text-zinc-500">
   422→              {draft.creator} · {draft.channel} · {draft.kind} · v{draft.version} ·{' '}
   423→              {timeAgo(draft.updatedAt)}
   424→            </p>
   425→          </div>
   426→        </div>
   427→        <div className="flex shrink-0 items-center gap-1.5">
   428→          {audit && approved ? (
   429→            <span
   430→              className="inline-flex items-center gap-1 rounded-md border border-emerald-200 bg-emerald-50 px-1.5 py-0.5 text-[11px] font-bold text-emerald-600"
   431→              title="Governed publication — tamper-evident certificate issued"
   432→            >
   433→              <BadgeCheck className="h-3 w-3" />
   434→              certified
   435→            </span>
   436→          ) : audit ? (
   437→            <span
   438→              className={cn(
   439→                'rounded-md border px-1.5 py-0.5 text-[11px] font-bold tabular-nums',
   440→                scoreTone
   441→              )}
   442→              title={`Latest audit score ${audit.score}/100`}
   443→            >
   444→              {audit.score}
   445→            </span>
   446→          ) : null}
   447→          <StatusChip status={draft.status} />
   448→        </div>
   449→      </div>
   450→
   451→      {/* rail */}
   452→      <div className="mt-4 px-1">
   453→        <div className="relative flex items-start justify-between">
   454→          {/* track */}
   455→          <span className="absolute left-6 right-6 top-[9px] h-[3px] rounded-full bg-zinc-200/80" />
   456→          <motion.span
   457→            className="absolute left-6 right-6 top-[9px] h-[3px] origin-left rounded-full bg-gradient-to-r from-emerald-500 to-teal-500"
   458→            initial={{ scaleX: 0 }}
   459→            animate={{ scaleX: fillPct / 100 }}
   460→            transition={{ delay: 0.25 + 0.06 * index, duration: 0.8, ease: [0.22, 1, 0.36, 1] }}
   461→          />
   462→          {stages.map((s, i) => {
   463→            const isActive = i === activeIdx
   464→            return (
   465→              <div key={s.id} className="relative z-10 flex w-12 flex-col items-center">
   466→                <span
   467→                  className={cn(
   468→                    'relative flex h-5 w-5 items-center justify-center rounded-full border-2 text-[9px] font-bold transition-all duration-300',
   469→                    s.done
   470→                      ? 'border-emerald-500 bg-emerald-500 text-white'
   471→                      : isActive
   472→                        ? 'border-emerald-400 bg-white text-emerald-500 shadow-[0_0_0_4px_rgba(16,185,129,0.18)]'
   473→                        : 'border-zinc-300 bg-zinc-50 text-zinc-400'
   474→                  )}
   475→                >
   476→                  {isActive && (
   477→                  <span
   478→                    aria-hidden
   479→                    className="absolute inset-0 animate-ping rounded-full bg-emerald-400/40"
   480→                  />
   481→                )}
   482→                {s.done ? <Check className="h-3 w-3" strokeWidth={3.5} /> : i + 1}
   483→                </span>
   484→                <span
   485→                  className={cn(
   486→                    'mt-1.5 text-[10px] leading-tight',
   487→                    s.done
   488→                      ? 'font-semibold text-zinc-700'
   489→                      : isActive
   490→                        ? 'font-bold text-emerald-600'
   491→                        : 'text-zinc-400'
   492→                  )}
   493→                >
   494→                  {s.label}
   495→                </span>
   496→              </div>

   498→          })}
   499→        </div>
   500→      </div>
   501→
   502→      {/* footer row */}
   503→      <div className="mt-3.5 flex items-center justify-between gap-3 border-t border-zinc-100 pt-3">
   504→        <p className="min-w-0 truncate text-xs text-zinc-500">
   505→          {allDone ? (
   506→            <>
   507→              <span className="font-semibold text-emerald-600">Done:</span>{' '}
   508→              approved & certified — view the tamper-evident certificate
   509→            </>
   510→          ) : (
   511→            <>
   512→              <span className="font-semibold text-zinc-700">Next:</span>{' '}
   513→              {NEXT_HINT[nextStage!.id]}{' '}
   514→              <span className="text-zinc-400">· {VIEW_LABEL[route]}</span>

   835→function describeEvent(action: string, details: Record<string, unknown>): string {
   836→  switch (action) {
   837→    case 'AUDIT_RUN':
   838→      return `6-pillar audit completed — score ${details.score}, ${details.findings} findings (${details.suppressed} suppressed)`
   839→    case 'AI_CALL':
   840→      return `AI call: ${String(details.purpose ?? 'inference').slice(0, 70)}`
   841→    case 'OVERRIDE':
   842→      return `Human override on ${details.findingKey} — “${String(details.justification ?? '').slice(0, 48)}…”`
   843→    case 'STATUS_CHANGE':
   844→      return `Status: ${details.from ?? '∅'} → ${details.to}`
   845→    case 'CERT_ISSUED':
   846→      return 'Compliance certificate issued'
   847→    case 'VAULT_INIT':
   848→      return String(details.message ?? 'Vault initialised')
   849→    case 'VAULT_EDIT':
   850→      return `Vault clause ${details.verb} — ${details.clauseId}`
   851→    case 'ACCEPT':
   852→      return `Finding ${details.findingKey} accepted`
   853→    default:
   854→      return action
   855→  }
