   146→   146→function DraftPanel({
   147→   147→  draft,
   148→   148→  audit,
   149→   149→  onAuditDone,
   150→   150→  actor,
   151→   151→}: {
   152→   152→  draft: DraftDTO
   153→   153→  audit: AuditResult | null
   154→   154→  onAuditDone: () => Promise<void>
   155→   155→  actor: string
   156→   156→}) {
   157→   157→  const { selectDraft, setView } = useApp()
   158→   158→  const [running, setRunning] = useState(false)
   159→   159→  const [stage, setStage] = useState(0) // 0 idle, 1..6 pillar animation
   160→   160→  const [result, setResult] = useState<AuditResult | null>(audit)
   161→   161→  const [content, setContent] = useState(draft.content)
   162→   162→  const [altText, setAltText] = useState(draft.imageMeta?.altText ?? '')
   163→   163→  const [showRedacted, setShowRedacted] = useState(false)
   164→   164→  const runToken = useRef(0)
   165→   165→
   166→   166→  useEffect(() => {
   167→   167→    setResult(audit)
   168→   168→    setContent(draft.content)
   169→   169→    setAltText(draft.imageMeta?.altText ?? '')
   170→   170→  }, [draft.id, draft.version, draft.content, audit])
   171→   171→
   172→   172→  const dirty =
   173→   173→    draft.kind === 'text' ? content !== draft.content : altText !== (draft.imageMeta?.altText ?? '')
   174→   174→
   175→   175→  const saveEdits = async () => {
   176→   176→    const res = await fetch(`/api/drafts/${draft.id}`, {
   177→   177→      method: 'PATCH',
   178→   178→      headers: { 'Content-Type': 'application/json' },
   179→   179→      body: JSON.stringify(draft.kind === 'text' ? { content, actor } : { altText, actor }),
   180→   180→    })
   181→   181→    if (!res.ok) {
   182→   182→      toast({ title: 'Save failed', description: (await res.json()).error, variant: 'destructive' })
   183→   183→      return false
   184→   184→    }
   185→   185→    await onAuditDone()
   186→   186→    toast({ title: 'Draft saved', description: 'Content changed — re-run the audit before publishing.' })
   187→   187→    return true
   188→   188→  }
   189→   189→
   190→   190→  const runAudit = async () => {
   191→   191→    if (dirty) {
   192→   192→      const ok = await saveEdits()
   193→   193→      if (!ok) return
   194→   194→    }
   195→   195→    setRunning(true)
   196→   196→    setResult(null)
   197→   197→    setStage(1)
   198→   198→    const token = ++runToken.current
   199→   199→
   200→   200→    // staged pillar animation while the engine runs
   201→   201→    const timer = setInterval(() => {
   202→   202→      setStage((s) => (s < 6 ? s + 1 : s))
   203→   203→    }, 500)
   204→   204→
   205→   205→    try {
   206→   206→      const res = await fetch('/api/audit', {
   207→   207→        method: 'POST',
   208→   208→        headers: { 'Content-Type': 'application/json' },
   209→   209→        body: JSON.stringify({ draftId: draft.id, actor }),
   210→   210→      })
   211→   211→      const body = await res.json()
   212→   212→      if (!res.ok) throw new Error(body.error ?? 'Audit failed')
   213→   213→      if (token !== runToken.current) return
   214→   214→      clearInterval(timer)
   215→   215→      setStage(6)
   216→   216→      // small delay so the last pillar animation completes
   217→   217→      await new Promise((r) => setTimeout(r, 450))
   218→   218→      setResult(body.audit as AuditResult)
   219→   219→      await onAuditDone()
   220→   220→      toast({
   221→   221→        title: `Audit complete — compliance score ${body.audit.score}/100`,
   222→   222→        description: `${body.audit.findings.length} findings · ${body.audit.suppressed} ungrounded AI flags suppressed`,
   223→   223→      })
   224→   224→    } catch (err) {
   225→   225→      clearInterval(timer)
   226→   226→      toast({
   227→   227→        title: 'Audit failed',
   228→   228→        description: err instanceof Error ? err.message : 'Unknown error',
   229→   229→        variant: 'destructive',
   230→   230→      })
   231→   231→    } finally {
   232→   232→      if (token === runToken.current) {
   233→   233→        setRunning(false)
   234→   234→        setStage(0)
   235→   235→      }
   236→   236→    }
   237→   237→  }
   238→   238→
   239→   239→  const editable = ['DRAFT', 'CHANGES_REQUESTED', 'AUDITED'].includes(draft.status)
   240→   240→
   241→   241→  return (
   242→   242→    <div className="space-y-4">
   243→   243→      {/* header */}
   244→   244→      <div className="rounded-xl border border-zinc-200 bg-white p-4 shadow-sm">
   245→   245→        <div className="flex flex-wrap items-start justify-between gap-3">
   246→   246→          <div className="min-w-0">
   247→   247→            <div className="flex flex-wrap items-center gap-2">
   248→   248→              <StatusChip status={draft.status} />
   249→   249→              <span className="text-[11px] font-semibold text-zinc-500">
   250→   250→                {draft.channel} · v{draft.version} · {draft.creator}
   251→   251→              </span>
   252→   252→            </div>
   253→   253→            <h1 className="mt-1.5 text-xl font-bold tracking-tight text-zinc-900">
   254→   254→              {draft.title}
   255→   255→            </h1>
   256→   256→          </div>
   257→   257→          <div className="flex items-center gap-2">
   258→   258→            {dirty && editable && (
   259→   259→              <Button variant="outline" size="sm" onClick={saveEdits}>
   260→   260→                Save draft
   261→   261→              </Button>
   262→   262→            )}
   263→   263→            <Button
   264→   264→              onClick={runAudit}
   265→   265→              disabled={running || draft.status === 'APPROVED'}
   266→   266→              className="bg-emerald-600 hover:bg-emerald-500"
   267→   267→            >
   268→   268→              {running ? (
   269→   269→                <>
   270→   270→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Auditing…
   271→   271→                </>
   272→   272→              ) : (
   273→   273→                <>
   274→   274→                  <ScanSearch className="mr-2 h-4 w-4" /> Run 6-Pillar Audit
   275→   275→                </>
   276→   276→              )}
   277→   277→            </Button>
   278→   278→          </div>
   279→   279→        </div>
   280→   280→      </div>
   281→   281→
   282→   282→      {/* content editor */}
   283→   283→      {draft.kind === 'text' ? (
   284→   284→        <div className="rounded-xl border border-zinc-200 bg-white p-4 shadow-sm">
   285→   285→          <div className="mb-2 flex items-center justify-between">
   286→   286→            <Label className="text-xs font-semibold uppercase tracking-wider text-zinc-500">
   287→   287→              Draft copy
   288→   288→            </Label>
   289→   289→            <span className="text-[11px] text-zinc-400">
   290→   290→              {content.trim().split(/\s+/).filter(Boolean).length} words
   291→   291→            </span>
   292→   292→          </div>
   293→   293→          <Textarea
   294→   294→            value={content}
   295→   295→            onChange={(e) => setContent(e.target.value)}
   296→   296→            disabled={!editable || running}
   297→   297→            className="min-h-[220px] font-mono text-[13px] leading-relaxed"
   298→   298→            placeholder="Paste the campaign copy…"
   299→   299→          />
   300→   300→        </div>
   301→   301→      ) : (
   302→   302→        <div className="rounded-xl border border-zinc-200 bg-white p-4 shadow-sm">
   303→   303→          <div className="mb-2 flex items-center justify-between">
   304→   304→            <Label className="text-xs font-semibold uppercase tracking-wider text-zinc-500">
   305→   305→              Promotional asset
   306→   306→            </Label>
   307→   307→            <span className="text-[11px] text-zinc-400">
   308→   308→              {draft.imageMeta?.width}×{draft.imageMeta?.height}px
   309→   309→            </span>
   310→   310→          </div>
   311→   311→          <div className="grid gap-4 sm:grid-cols-[minmax(0,320px)_1fr]">
   312→   312→            {draft.imageData && (
   313→   313→              <img
   314→   314→                src={draft.imageData}
   315→   315→                alt={draft.title}
   316→   316→                className="w-full rounded-lg border border-zinc-200 object-contain"
   317→   317→              />
   318→   318→            )}
   319→   319→            <div className="space-y-3">
   320→   320→              <div>
   321→   321→                <p className="text-[11px] font-semibold uppercase tracking-wider text-zinc-500">
   322→   322→                  Extracted palette (client-side, pre-upload)
   323→   323→                </p>
   324→   324→                <div className="mt-1.5 flex flex-wrap gap-1.5">
   325→   325→                  {(draft.imageMeta?.palette ?? []).map((s) => (
   326→   326→                    <span
   327→   327→                      key={s.hex}
   328→   328→                      className="inline-flex items-center gap-1.5 rounded-md border border-zinc-200 bg-zinc-50 px-2 py-1 font-mono text-[11px] font-semibold text-zinc-700"
   329→   329→                    >
   330→   330→                      <HexSwatch hex={s.hex} size={14} />
   331→   331→                      {s.hex} · {(s.ratio * 100).toFixed(0)}%
   332→   332→                    </span>
   333→   333→                  ))}
   334→   334→                </div>
   335→   335→              </div>
   336→   336→              <div>
   337→   337→                <Label className="text-xs">Alt text {altText ? '' : '(missing — will be flagged)'}</Label>
   338→   338→                <Input
   339→   339→                  className="mt-1 text-xs"
   340→   340→                  value={altText}
   341→   341→                  onChange={(e) => setAltText(e.target.value)}
   342→   342→                  disabled={!editable || running}
   343→   343→                  placeholder="Describe the image for screen readers…"
   344→   344→                />
   345→   345→              </div>
   346→   346→            </div>
   347→   347→          </div>
   348→   348→        </div>
   349→   349→      )}
   350→   350→
   351→   351→      {/* live pillar animation */}
   352→   352→      <AnimatePresence>
   353→   353→        {running && <PillarProgress stage={stage} kind={draft.kind} />}
   354→   354→      </AnimatePresence>
   355→   355→
   356→   356→      {/* results */}
   357→   357→      {result && (
   358→   358→        <AuditResultPanel
   359→   359→          result={result}
   360→   360→          draft={draft}
   361→   361→          onGoRemediate={() => {
   362→   362→            selectDraft(draft.id, 'remediation')
   363→   363→            setView('remediation')
   364→   364→          }}
   365→   365→          showRedacted={showRedacted}
   366→   366→          onToggleRedacted={() => setShowRedacted((s) => !s)}
   367→   367→        />
   368→   368→      )}
   369→   369→    </div>
   370→   370→  )
   371→   371→}

   373→   373→function PillarProgress({ stage, kind }: { stage: number; kind: string }) {
   374→   374→  const pillars: Pillar[] = ['tone', 'terminology', 'claims', 'visual', 'accessibility', 'confidentiality']
   375→   375→  return (
   376→   376→    <motion.div
   377→   377→      initial={{ opacity: 0, y: 8 }}
   378→   378→      animate={{ opacity: 1, y: 0 }}
   379→   379→      exit={{ opacity: 0 }}
   380→   380→      className="rounded-xl border border-emerald-200 bg-emerald-50/40 p-4"
   381→   381→    >
   382→   382→      <p className="flex items-center gap-2 text-xs font-semibold text-emerald-800">
   383→   383→        <Loader2 className="h-4 w-4 animate-spin" />
   384→   384→        Parallel 6-pillar engine running — deterministic math & citation-gated RAG…
   385→   385→      </p>
   386→   386→      <div className="mt-3 grid grid-cols-2 gap-2 sm:grid-cols-3 lg:grid-cols-6">
   387→   387→        {pillars.map((p, i) => {
   388→   388→          const state = i < stage ? 'running' : 'queued'
   389→   389→          return (
   390→   390→            <div
   391→   391→              key={p}
   392→   392→              className={cn(
   393→   393→                'flex items-center gap-2 rounded-lg border px-2.5 py-2 text-[11px] font-semibold',
   394→   394→                state === 'running'
   395→   395→                  ? 'border-emerald-300 bg-white text-emerald-800'
   396→   396→                  : 'border-zinc-200 bg-white/60 text-zinc-400'
   397→   397→              )}
   398→   398→            >
   399→   399→              {state === 'running' ? (
   400→   400→                <Loader2 className="h-3.5 w-3.5 shrink-0 animate-spin text-emerald-500" />
   401→   401→              ) : (
   402→   402→                <CircleDot className="h-3.5 w-3.5 shrink-0 text-zinc-300" />
   403→   403→              )}
   404→   404→              {PILLAR_META[p].short}
   405→   405→              {kind === 'text' && ['visual', 'accessibility'].includes(p) && (
   406→   406→                <span className="text-[9px] font-normal text-zinc-400">(n/a)</span>
   407→   407→              )}
   408→   408→            </div>
   409→   409→          )
   410→   410→        })}
   411→   411→      </div>
   412→   412→    </motion.div>
   413→   413→  )
   414→   414→}

   418→   418→function AuditResultPanel({
   419→   419→  result,
   420→   420→  draft,
   421→   421→  onGoRemediate,
   422→   422→  showRedacted,
   423→   423→  onToggleRedacted,
   424→   424→}: {
   425→   425→  result: AuditResult
   426→   426→  draft: DraftDTO
   427→   427→  onGoRemediate: () => void
   428→   428→  showRedacted: boolean
   429→   429→  onToggleRedacted: () => void
   430→   430→}) {
   431→   431→  const verdict =
   432→   432→    result.score >= 85
   433→   433→      ? { text: 'Ready to publish', cls: 'text-emerald-600' }
   434→   434→      : result.score >= 60
   435→   435→        ? { text: 'Minor fixes recommended', cls: 'text-amber-600' }
   436→   436→        : { text: 'Must not ship as-is', cls: 'text-rose-600' }
   437→   437→
   438→   438→  return (
   439→   439→    <motion.div
   440→   440→      initial={{ opacity: 0, y: 12 }}
   441→   441→      animate={{ opacity: 1, y: 0 }}
   442→   442→      className="space-y-4"
   443→   443→    >
   444→   444→      {/* score + pillars */}
   445→   445→      <div className="rounded-xl border border-zinc-200 bg-white p-5 shadow-sm">
   446→   446→        <div className="flex flex-wrap items-center gap-6">
   447→   447→          <div className="flex items-center gap-5">
   448→   448→            <ScoreRing score={result.score} />
   449→   449→            <div>
   450→   450→              <p className="text-[11px] font-semibold uppercase tracking-wider text-zinc-400">
   451→   451→                Compliance verdict
   452→   452→              </p>
   453→   453→              <p className={cn('text-xl font-bold', verdict.cls)}>{verdict.text}</p>
   454→   454→              <p className="mt-1 text-xs text-zinc-500">
   455→   455→                {result.findings.length} findings ·{' '}
   456→   456→                {result.findings.filter((f) => f.severity === 'critical').length} critical
   457→   457→              </p>
   458→   458→              <div className="mt-2 flex flex-wrap items-center gap-1.5">
   459→   459→                <span className="inline-flex items-center gap-1 rounded-md border border-emerald-200 bg-emerald-50 px-2 py-1 text-[10.5px] font-semibold text-emerald-700">
   460→   460→                  <ShieldCheck className="h-3 w-3" />
   461→   461→                  Zero-hallucination guard: {result.suppressed} ungrounded AI flag
   462→   462→                  {result.suppressed === 1 ? '' : 's'} suppressed
   463→   463→                </span>
   464→   464→                {result.aiLedger.map((l, i) => (
